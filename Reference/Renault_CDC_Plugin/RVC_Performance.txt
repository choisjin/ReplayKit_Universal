    [Arguments]    ${sleep_delay}    ${wakeup_delay}    ${delay}

    IF    '${scar_mode}' == 'API'
        Log With Time    Use SCAR API path
        Log With Time    Send UTC time event in SCAR by using API
        Send request in SCAR by using API     ${UTC_TIME_START_URL}       ${UTC_TIME_START_HEADERS}       ${UTC_TIME_START_DATA}
        Wait    5
        Send request in SCAR by using API     ${UTC_TIME_URL}       ${UTC_TIME_HEADERS}       ${UTC_TIME_DATA}
        Wait    5
        Send request in SCAR by using API     ${REAQUEST_START_URL}       ${REAQUEST_START_HEADERS}       ${REAQUEST_START_DATA}
        Wait    5

        Log With Time   Send pnp off switch in SCAR by using API
        Send request in SCAR by using API     ${PNP_SWITCH_OFF_URL}       ${PNP_SWITCH_OFF_HEADERS}       ${PNP_SWITCH_OFF_DATA}
        Wait    10
        Log With Time    Switch Wakeup to I
            RELAY Send String    cdc_wk_off

        ${start_time}=    Get Time    epoch
        ${rc}    ${msg}=    Run Keyword And Ignore Error
        ...    Wait Until Keyword Succeeds
        ...    ${sleep_delay}
        ...    5s
        ...    Wait For Serial String
        ...    PS:SHUTDOWN_DONE->OFF. ACT_Done:0x00000040
        ...    MCU

        ${end_time}=    Get Time    epoch
        ${elapsed}=     Evaluate    round(${end_time} - ${start_time}, 1)

        IF    '${rc}' == 'PASS'
            Log With Time    Sleep entry detected after ${elapsed} seconds
        ELSE
            Log    Sleep entry NOT detected within ${sleep_delay}s (waited ${elapsed}s)    WARN
        END

        Arduino Send String    RVC_START

        Log With Time   Send pnp on switch in SCAR by using API
        Send request in SCAR by using API     ${PNP_SWITCH_ON_URL}       ${PNP_SWITCH_ON_HEADERS}       ${PNP_SWITCH_ON_DATA}
        # ECU Sleep Delay
        Wait Ms    100
    ELSE IF    '${scar_mode}' == 'DOCKER'
        Log With Time    Use SCAR docker command path
        Exec In SCAR Container
            ...    cd /home/scar && python3 scripts/test/start.py -s VehicleUtcTimeSvcProv --ends 2025_r10 -e PCU_PROXY_FrontEnd_PIU_Mst
            Wait    3
            Exec In SCAR Container
            ...    cd /home/scar && python3 scripts/test/write.py -s VehicleUtcTimeSvcProv --ends 2025_r10 -m VehicleUtcTimeEvent -d time:now -d vehicleUtcTimeValueState:VALUE_STATE_VALID
            Wait    3
            Exec In SCAR Container
            ...    cd /home/scar && python3 scripts/test/write.py -s VehicleUtcTimeSvcProv --ends 2025_r10 -m VehicleUtcTimeEvent -d time:now -d vehicleUtcTime.vehicleUtcTimeValidity:VEHICLE_UTC_TIME_VALIDITY_T_VALID
            Wait    3
            Exec In SCAR Container
            ...    cd /home/scar && python3 scripts/test/start.py -s RequestVehicleUtcTimeSvcProv --ends 2025_r10 -e PCU_PROXY_FrontEnd_PIU_Mst
            Wait    3
            Log With Time   Send pnp off switch in SCAR by using command
            Exec In SCAR Container
            ...    cd /home/scar && ENDS=2025_r10 ./scripts/test_sequences/pnp/sleep_power_sequence.sh off --frontend
            Wait    3
            Log With Time    Switch Wakeup to I
            RELAY Send String    cdc_wk_off
#            RELAY Send String    all_slp

            Log With Time    Wait until MCU sleep entry detected

            ${start_time}=    Get Time    epoch
            ${rc}    ${msg}=    Run Keyword And Ignore Error
            ...    Wait Until Keyword Succeeds
            ...    ${sleep_delay}
            ...    2s
            ...    Wait For Serial String
            ...    PS:SHUTDOWN_DONE->OFF. ACT_Done:0x00000040
            ...    MCU

            ${end_time}=    Get Time    epoch
            ${elapsed}=     Evaluate    round(${end_time} - ${start_time}, 1)

            IF    '${rc}' == 'PASS'
                Log With Time    Sleep entry detected after ${elapsed} seconds
            ELSE
                Log    Sleep entry NOT detected within ${sleep_delay}s (waited ${elapsed}s)    WARN
            END
            Arduino Send String    RVC_START
            Log With Time   Send pnp on switch in SCAR by using command
            Exec In SCAR Container
            ...    cd /home/scar && ENDS=2025_r10 ./scripts/test_sequences/pnp/sleep_power_sequence.sh on --frontend
            # ECU Sleep Delay
            Wait Ms    100
    ELSE
        Log    SCAR not available. Skip SCAR control part.    WARN
    END
    Log With Time    Switch Wakeup to II
    RELAY Send String    cdc_wk_on
#    RELAY Send String    all_wk on
    Wait    3
    # RVC_On
    Send Signal To Test Harness    ${TH_CLIENT_DIR}    some_ip_broker_com_sdv_ampere_common_fnd_sdv_energy_power_distribution_vehicle_power_mode_unique    /home/cdc/0.scripts/runJSON/PowerOn.json

    Send Signal To Test Harness    ${TH_CLIENT_DIR}    some_ip_broker_com_sdv_ampere_common_fnd_sdv_vehicle_dynamics_speed_info_timestamped_unique    /home/cdc/0.scripts/runJSON/SpeedInfoTimestampedRVC.json

        ...    ${TH_CLIENT_DIR}
        ...    some_ip_broker_com_sdv_ampere_common_fnd_sdv_powertrain_gear_lever_position_accepted_unique
        ...    /home/cdc/0.scripts/runJSON/com_sdv_ampere_common_fnd_sdv_powertrain_gear_lever_PositionAccepted.json
        ...    ${TH_ADDR}

    # RVC_Off
    Send Signal To Test Harness    ${TH_CLIENT_DIR}    some_ip_broker_com_sdv_ampere_common_fnd_sdv_powertrain_gear_lever_position_accepted_unique    /home/cdc/0.scripts/runJSON/GearNeutral.json
    Wait     10

    ${status}    ${result}=    Run Keyword And Ignore Error    ADB Result Check    getprop vendor.alliance.boot.reason    quick    ${AVN_AND_1}

        Log With Time    FVM Qucik Boot pass : ${result}[1]
    ELSE
        Log    FVM Quick Boot fail : ${result}[1]    WARN
    END

    ${status}    ${result}=    Run Keyword And Ignore Error    ADB Result Check    getprop vendor.boot.vm_boot_reason    quick    ${AVN_AND_2}
